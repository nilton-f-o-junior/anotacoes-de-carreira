## Operadores Matemáticos

```javascript
// +
let num_a = 1;
let num_b = 2;

let result = num_a + num_b;
console.log(result);

// -
let num_c = 1;
let num_d = 2;

let result = num_c - num_d;
console.log(result);

// *
let num_e = 1;
let num_f = 2;

let result = num_e * num_f;
console.log(result);

// /
let num_g = 1;
let num_h = 2;

let result = num_g / num_h;
console.log(result);

// %
let num_i = 1;
let num_j = 2;

let result = num_i % num_j;
console.log(result);

// **
let num_k = 1;
let num_l = 2;

let result = num_k ** num_l;
console.log(result);
```

## Operadores de Atribuição

```javascript
let num_a = 2;
let num_b = 2;

let result = num_a + num_b;
console.log(result);

// +=
result += num_a;
console.log(result);

// -=
result -= num_a;
console.log(result);

// *=
result *= num_a;
console.log(result);

// /=
result /= num_a;
console.log(result);

// %=
result %= num_a;
console.log(result);

// **=
result **= num_a;
console.log(result);
```

## Operadores de Comparação

```javascript
let num_a = 1;
let num_b = 2;

// > (maior)
console.log(num_a > num_b);

// < (menor)
console.log(num_a < num_b);

// >= (maior ou igual)
console.log(num_a >= num_b);

// <= (menor ou igual)
console.log(num_a <= num_b);

// == (igual)
console.log(num_a == num_b);

// != (diferente)
console.log(num_a != num_b);

// === (igual)
console.log(num_a === num_b);

// !== (diferente)
console.log(num_a !== num_b);
```
